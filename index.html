3. Operators: sum, product, average

<html>
<head>
<title>operators</title>

<script type="text/javascript">
let n1=Number(prompt("Enter number 1"));
let n2=Number(prompt("Enter number 2"));
let n3=Number(prompt("Enter number 3"));

function sum(){
let sum= n1+n2+n3;
alert("sum is:" + sum);
}

function product(){
let product= n1*n2*n3;
alert("product is:" + product);
}

function avg(){
let avg= (n1+n2+n3)/3;
alert("avg is:" + avg);
}

</script>
</head>

<body>
<input type="button" onclick="sum()" value="Sum">
<input type="button" onclick="product()" value="Product">
<input type="button" onclick="avg()" value="Average">
</body>

8. Factorial

<html>
<body>
<script type="text/javascript">

let a, i, fact = 1;
a= Number(prompt("Enter the number you want factorial of"));

fact = fact * i;

}

document.write("factorial of"+ a +"is"+ fact)

</script>
</body>
</html>
